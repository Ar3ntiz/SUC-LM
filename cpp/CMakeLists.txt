# SUC C++ runtime — milestone 1: CPU SUC1 reader + reference decoder (no external dependencies);
# milestone 2 (Apple only): Metal backend for one SUC linear layer (metal-cpp, kernels compiled at run time).
cmake_minimum_required(VERSION 3.21)
project(suc_runtime LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Bit-exactness with the Python reference decoder requires that the compiler never contracts a*b+c
# into an fma (numpy rounds the product and the sum separately); the explicit std::fma in the
# Hadamard H_m step is the only fused operation.
add_compile_options(-ffp-contract=off -Wall -Wextra)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  include(CheckCXXCompilerFlag)
  check_cxx_compiler_flag("-march=armv8-a+crypto" SUC_HAS_ARM_CRYPTO)
  if(SUC_HAS_ARM_CRYPTO AND NOT APPLE)   # Apple clang enables the SHA-2 extension by default
    add_compile_options(-march=armv8-a+crypto)
  endif()
endif()

find_package(Threads REQUIRED)

add_library(suc STATIC
  src/bits.cpp
  src/decode.cpp
  src/hadamard.cpp
  src/json.cpp
  src/npy.cpp
  src/numpy_random.cpp
  src/parallel.cpp
  src/sampling.cpp
  src/sha256.cpp
  src/suc1.cpp
  src/tokenizer.cpp)
target_include_directories(suc PUBLIC include)
target_link_libraries(suc PUBLIC Threads::Threads)

add_executable(suc1-inspect tools/suc1_inspect.cpp)
target_link_libraries(suc1-inspect PRIVATE suc)
add_executable(suc1-decode tools/suc1_decode.cpp)
target_link_libraries(suc1-decode PRIVATE suc)
add_executable(suc-tokenize tools/suc_tokenize.cpp)
target_link_libraries(suc-tokenize PRIVATE suc)

add_executable(suc-sampling-bench tools/suc_sampling_bench.cpp)
target_link_libraries(suc-sampling-bench PRIVATE suc)

enable_testing()
add_executable(suc-prefix-tests tests/test_live_prefix.cpp)
target_include_directories(suc-prefix-tests PRIVATE include)
add_test(NAME suc-prefix-tests COMMAND suc-prefix-tests)
add_executable(suc-tests tests/test_main.cpp)
target_link_libraries(suc-tests PRIVATE suc)
add_test(NAME suc-tests COMMAND suc-tests)

# Audit F10: int64 boundaries of the request helpers, built with -fsanitize=float-cast-overflow (non-recoverable) when
# the toolchain supports it, so an out-of-range double-to-int64 conversion aborts the test.
include(CheckCXXSourceCompiles)
set(SUC_UBSAN_FLAGS -fsanitize=float-cast-overflow -fno-sanitize-recover=float-cast-overflow)
set(CMAKE_REQUIRED_FLAGS "-fsanitize=float-cast-overflow -fno-sanitize-recover=float-cast-overflow")
set(CMAKE_REQUIRED_LINK_OPTIONS ${SUC_UBSAN_FLAGS})
check_cxx_source_compiles("int main() { return 0; }" SUC_HAS_FLOAT_CAST_SANITIZER)
unset(CMAKE_REQUIRED_FLAGS)
unset(CMAKE_REQUIRED_LINK_OPTIONS)
add_executable(suc-request-params-tests tests/test_request_params_ubsan.cpp src/json.cpp)
target_include_directories(suc-request-params-tests PRIVATE include)
if(SUC_HAS_FLOAT_CAST_SANITIZER)
  target_compile_options(suc-request-params-tests PRIVATE ${SUC_UBSAN_FLAGS})
  target_link_options(suc-request-params-tests PRIVATE ${SUC_UBSAN_FLAGS})
endif()
add_test(NAME suc-request-params-tests COMMAND suc-request-params-tests)

# ---- Metal backend (macOS): metal-cpp headers; cpp/metal/suc_kernels.metal is embedded as a string and compiled
# at run time (newLibraryWithSource), so no offline `metal` compiler is needed. Mirrors cpp/build.sh.
if(APPLE AND NOT SUC_NO_METAL)
  set(SUC_MSL "${CMAKE_CURRENT_SOURCE_DIR}/metal/suc_kernels.metal")
  set(SUC_MODEL_MSL "${CMAKE_CURRENT_SOURCE_DIR}/metal/suc_model_kernels.metal")
  set(SUC_QWEN4_MSL "${CMAKE_CURRENT_SOURCE_DIR}/metal/suc_qwen4_kernels.metal")
  set(SUC_DS4_MSL "${CMAKE_CURRENT_SOURCE_DIR}/metal/suc_ds4_kernels.metal")
  set(SUC_GLM5_MSL "${CMAKE_CURRENT_SOURCE_DIR}/metal/suc_glm5_kernels.metal")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SUC_MSL}" "${SUC_MODEL_MSL}" "${SUC_QWEN4_MSL}"
               "${SUC_DS4_MSL}" "${SUC_GLM5_MSL}")
  file(READ "${SUC_MSL}" SUC_MSL_TEXT)
  file(READ "${SUC_MODEL_MSL}" SUC_MODEL_MSL_TEXT)
  file(READ "${SUC_QWEN4_MSL}" SUC_QWEN4_MSL_TEXT)
  file(READ "${SUC_DS4_MSL}" SUC_DS4_MSL_TEXT)
  file(READ "${SUC_GLM5_MSL}" SUC_GLM5_MSL_TEXT)
  set(SUC_MSL_CPP "${CMAKE_CURRENT_BINARY_DIR}/gen/suc_kernels_msl.cpp")
  file(WRITE "${SUC_MSL_CPP}.tmp"
       "// Generated by CMake from cpp/metal/suc_kernels.metal, suc_model_kernels.metal and suc_qwen4_kernels.metal.\n"
       "namespace suc::metal {\n"
       "extern const char* const kEmbeddedKernelSource;\nextern const char* const kEmbeddedModelKernelSource;\n"
       "extern const char* const kEmbeddedQwen4KernelSource;\n"
       "extern const char* const kEmbeddedDs4KernelSource;\n"
       "extern const char* const kEmbeddedGlm5KernelSource;\n"
       "const char* const kEmbeddedKernelSource = R\"SUCMSL(${SUC_MSL_TEXT})SUCMSL\";\n"
       "const char* const kEmbeddedModelKernelSource = R\"SUCMSL(${SUC_MODEL_MSL_TEXT})SUCMSL\";\n"
       "const char* const kEmbeddedQwen4KernelSource = R\"SUCMSL(${SUC_QWEN4_MSL_TEXT})SUCMSL\";\n"
       "const char* const kEmbeddedDs4KernelSource = R\"SUCMSL(${SUC_DS4_MSL_TEXT})SUCMSL\";\n"
       "const char* const kEmbeddedGlm5KernelSource = R\"SUCMSL(${SUC_GLM5_MSL_TEXT})SUCMSL\";\n"
       "}  // namespace suc::metal\n")
  file(COPY_FILE "${SUC_MSL_CPP}.tmp" "${SUC_MSL_CPP}" ONLY_IF_DIFFERENT)
  add_library(suc_metal STATIC src/metal/metal.cpp src/metal/model.cpp src/metal/switch.cpp src/metal/qwen4.cpp src/metal/deepseek_v4.cpp src/metal/expert_mixed.cpp src/metal/expert_stream.cpp src/metal/glm5.cpp "${SUC_MSL_CPP}")
  target_include_directories(suc_metal SYSTEM PUBLIC third_party/metal-cpp)
  target_link_libraries(suc_metal PUBLIC suc "-framework Metal" "-framework Foundation" "-framework QuartzCore")
  add_executable(suc1-linear tools/suc1_linear.cpp)
  target_link_libraries(suc1-linear PRIVATE suc_metal)
  add_executable(suc1-generate tools/suc1_generate.cpp)
  target_link_libraries(suc1-generate PRIVATE suc_metal)
  add_executable(suc1-server tools/suc1_server.cpp)
  target_link_libraries(suc1-server PRIVATE suc_metal)
  add_executable(suc1-moe tools/suc1_moe.cpp)
  target_link_libraries(suc1-moe PRIVATE suc_metal)
  add_executable(suc-expert-bench tools/suc_expert_bench.cpp)
  target_link_libraries(suc-expert-bench PRIVATE suc_metal)
  add_executable(suc-linear-bench tools/suc_linear_bench.cpp)
  target_link_libraries(suc-linear-bench PRIVATE suc_metal)
  add_executable(suc-attention-bench tools/suc_attention_bench.cpp)
  target_link_libraries(suc-attention-bench PRIVATE suc_metal)
  add_executable(suc-mm-bench tools/suc_mm_bench.cpp)
  target_link_libraries(suc-mm-bench PRIVATE suc_metal)
  add_executable(suc-qwen-checkpoint-tests tests/test_qwen_checkpoint.cpp)
  target_include_directories(suc-qwen-checkpoint-tests PRIVATE tools)
  target_link_libraries(suc-qwen-checkpoint-tests PRIVATE suc_metal)
  add_executable(suc-metal-tests tests/test_metal.cpp)
  target_link_libraries(suc-metal-tests PRIVATE suc_metal)
  add_test(NAME suc-metal-tests COMMAND suc-metal-tests)
  # Python cross-check of the qwen4_exp graph against the Python/MLX SUC1 runtime (tiny random files; needs the project
  # venv at ../.venv). SUC_CPP_BUILD points the test at this build directory.
  set(SUC_VENV_PYTHON "${CMAKE_CURRENT_SOURCE_DIR}/../.venv/bin/python")
  if(EXISTS "${SUC_VENV_PYTHON}")
    add_test(NAME suc-qwen4-crosscheck
             COMMAND "${SUC_VENV_PYTHON}" -m pytest -q tests/unit/test_cpp_qwen4.py
             WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
    set_tests_properties(suc-qwen4-crosscheck PROPERTIES ENVIRONMENT "SUC_CPP_BUILD=${CMAKE_CURRENT_BINARY_DIR}")
    add_test(NAME suc-deepseek-v4-crosscheck
             COMMAND "${SUC_VENV_PYTHON}" -m pytest -q tests/unit/test_cpp_deepseek_v4.py
             WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
    set_tests_properties(suc-deepseek-v4-crosscheck PROPERTIES ENVIRONMENT "SUC_CPP_BUILD=${CMAKE_CURRENT_BINARY_DIR}")
    add_test(NAME suc-glm5-crosscheck
             COMMAND "${SUC_VENV_PYTHON}" -m pytest -q tests/unit/test_cpp_glm5.py
             WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
    set_tests_properties(suc-glm5-crosscheck PROPERTIES ENVIRONMENT "SUC_CPP_BUILD=${CMAKE_CURRENT_BINARY_DIR}")
    add_test(NAME suc-server-openai
             COMMAND "${SUC_VENV_PYTHON}" -m pytest -q tests/unit/test_cpp_server.py
             WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
    set_tests_properties(suc-server-openai PROPERTIES ENVIRONMENT "SUC_CPP_BUILD=${CMAKE_CURRENT_BINARY_DIR}")
  endif()
